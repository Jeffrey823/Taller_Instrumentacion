******************************************* Gráfico de Bode para cada Vin: *******************************************

datos_prom <- aggregate(Ganancia ~ `f_in factor (Hz)` + `Vin factor (pk)`, data = mediciones_proyecto_2, FUN = mean)
library(ggplot2)
ggplot(datos_prom, aes(x = `f_in factor (Hz)`, y = Ganancia, color = factor(`Vin factor (pk)`))) +  geom_line() +
  geom_point() +
  scale_x_log10() +
  labs(title = "Gráfico de Bode por Vin",
       x = "Frecuencia (Hz)",
       y = "Ganancia",
       color = "Vin (pk)") +
  theme_minimal()
